  -c 'au MyConfig' \
  -c 'source $MYVIMRC' -c 'source $MYVIMRC' \
  -c 'au MyConfig' \
  -c 'qa!'

    *         echo "reading"
MyConfig  BufWritePre
    *         echo "saving"
